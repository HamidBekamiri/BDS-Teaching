From results to an app: data, business and presentation layers

Turn the workshop results (themes, extracted facts, triples, embeddings) into a Streamlit app that researchers can use: explore themes, compare them, and ask questions with cited answers. Three small files, one job each.

The idea: three layers

An app is easier to build, test and change when each part has one job. Change one layer and the others do not notice: swap Streamlit for Gradio, files for a database, or Ollama for an API model.

Data layerdata.py

Loads and caches the results: papers, extracted facts, triples, embeddings. Knows nothing about pages or the LLM.

Business layerlogic.py

Plain Python functions that answer questions: theme profile, method gap, search, a grounded answer. Knows nothing about Streamlit.

Presentation layerapp.py

Tabs, widgets and charts. Only calls the business layer. Knows nothing about files or how answers are made.

app.py → logic.py → data.py → files in data/. Never the other way round.

Data layer: data.py

@st.cache_data loads a file once and reuses it on every click; @st.cache_resource keeps one copy of a model for all users. Embedding questions uses the same model S3 used, so questions and papers live in the same space.

data.py
"""Data layer: load and cache the results. Knows nothing about Streamlit pages or the LLM."""
from pathlib import Path

import numpy as np
import pandas as pd
import streamlit as st

DATA = Path(__file__).parent / "data"


@st.cache_data                       # load once, reuse on every click
def papers() -> pd.DataFrame:
    return pd.read_json(DATA / "papers.jsonl", lines=True)


@st.cache_data
def triples() -> pd.DataFrame:
    return pd.read_csv(DATA / "triples.csv")


@st.cache_data
def embeddings() -> np.ndarray:
    return np.load(DATA / "embeddings.npy")


@st.cache_resource                   # models are resources: one copy for all users
def encoder():
    from sentence_transformers import SentenceTransformer
    return SentenceTransformer("sentence-transformers/all-mpnet-base-v2")   # the model S3 used

Business layer: logic.py

Each function answers one question a user may have. answer() is RAG + KG: it retrieves the closest papers, puts their extracted facts into the prompt, and asks the local model to answer with citations.

logic.py
"""Business layer: answer questions with plain Python. Knows nothing about Streamlit or file paths."""
import numpy as np
import pandas as pd
from openai import OpenAI

import data

MODEL, OLLAMA = "gemma4:e2b", "http://localhost:11434/v1"


def themes() -> list[str]:
    return sorted(data.papers().theme.unique())


def growth() -> pd.DataFrame:
    p = data.papers()
    return pd.crosstab(p.year, p.theme)


def theme_profile(theme: str, n: int = 8) -> dict:
    p, t = data.papers(), data.triples()
    ids = set(p.loc[p.theme == theme, "eid"])
    sub = t[t.paper_id.isin(ids)]
    top = lambda rel: sub[sub.relation == rel].object.value_counts().head(n)
    return {"papers": len(ids), "methods": top("USES_METHOD"), "theories": top("USES_THEORY"),
            "data": top("USES_DATA")}


def method_gap(a: str, b: str, n: int = 5) -> pd.Series:
    """Methods used in theme a but never in theme b."""
    ma, mb = theme_profile(a, 50)["methods"], theme_profile(b, 50)["methods"]
    return ma[~ma.index.isin(mb.index)].head(n)


def search(question: str, k: int = 6) -> pd.DataFrame:
    q = data.encoder().encode([question], normalize_embeddings=True)[0]
    scores = data.embeddings() @ q
    top = np.argsort(-scores)[:k]
    return data.papers().iloc[top].assign(score=scores[top])


def answer(question: str, k: int = 6) -> tuple[str, pd.DataFrame]:
    hits = search(question, k)
    sources = "\n".join(
        f"[{i}] {r.title} ({r.year}; {r.theme}) asks: {r.research_question} | "
        f"methods: {', '.join(r.methods)} | finds: {r.results}"
        for i, r in enumerate(hits.itertuples(), 1))
    client = OpenAI(base_url=OLLAMA, api_key="ollama", timeout=120)
    reply = client.chat.completions.create(
        model=MODEL, temperature=0, max_tokens=600, reasoning_effort="none",
        messages=[{"role": "system", "content": "Answer only from the numbered sources and cite them like [2]. "
                                                "If they do not answer the question, say so."},
                  {"role": "user", "content": f"Question: {question}\n\nSources:\n{sources}"}])
    return reply.choices[0].message.content, hits

Presentation layer: app.py

Four tabs, each a few lines that call one business function: Overview (growth), Explore a theme (methods, theories, data), Compare themes (method gaps) and Ask (chat with sources).

app.py
"""Presentation layer: pages, widgets and charts. Only calls the business layer."""
import streamlit as st

import logic

st.set_page_config(page_title="GenAI-in-business research", layout="wide")
st.title("GenAI-in-business research assistant")
overview, explore, compare, ask = st.tabs(["Overview", "Explore a theme", "Compare themes", "Ask"])

with overview:
    st.subheader("Papers per theme and year")
    st.line_chart(logic.growth())

with explore:
    theme = st.selectbox("Theme", logic.themes())
    prof = logic.theme_profile(theme)
    st.metric("Papers", prof["papers"])
    c1, c2, c3 = st.columns(3)
    c1.write("**Methods**");  c1.bar_chart(prof["methods"])
    c2.write("**Theories**"); c2.bar_chart(prof["theories"])
    c3.write("**Data**");     c3.bar_chart(prof["data"])

with compare:
    a, b = st.columns(2)
    ta = a.selectbox("Theme A", logic.themes(), key="a")
    tb = b.selectbox("Theme B", logic.themes(), index=1, key="b")
    st.write(f"Methods used in **{ta}** but never in **{tb}**:")
    st.dataframe(logic.method_gap(ta, tb))

with ask:
    q = st.chat_input("Ask about the research…")
    if q:
        st.chat_message("user").write(q)
        with st.spinner("Searching and reading…"):
            text, hits = logic.answer(q)
        st.chat_message("assistant").write(text)
        with st.expander("Sources"):
            st.dataframe(hits[["title", "year", "theme", "score"]])

Run it

The Ask tab needs Ollama running with gemma4:e2b (as in the workshop). The other tabs work without it.

6

Share it

Streamlit Community Cloud or Hugging Face Spaces host the app for free from a GitHub repository with the app folder and a requirements.txt. They cannot run Ollama: either point answer() at an API model (one change, in the business layer), or share only the first three tabs.

Only share data you may share, and never put an API key in the code: use the host's secrets settings.
